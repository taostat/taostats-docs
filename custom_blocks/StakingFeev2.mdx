import React, { useState } from 'react';

export const StakingFee2 = () => {
    const [subnet, setSubnet] = useState('');
  	const [fee, setFee] = useState('');
  	const [error, setError] = useState(null);

  const GetFeeFromAPI = async () => {
			const subnetValue = parseFloat(subnet);

      if (subnetValue) {
        try {
          const response = await fetch(`https://docsdemo.taostats.io/api/http/api/subnet/latest/v1?netuid=${subnetValue}`);
          if (!response.ok) {
 					     // Throws an error with the HTTP status text (e.g., "Not Found")
      					throw new Error(`HTTP error! Status: ${response.status}`);
    			}
					const json = await response.json();
          const staking_fee = json['data'][0]['fee_rate'];
					console.log(json);
          setFee(Math.round(staking_fee*100000)/100000*100);
          setError(null);
      	} catch (error) {
        	setError(error.message);
      		}
    	} else {
      	setError('Please enter a valid number');
    	};
  };
    const handleSubnetChange = (e) => {
      setSubnet(e.target.value);
    };

    return (
      <div class="my-box">
        <form onSubmit={(e) => e.preventDefault()}>
          <label>
            Subnet:
            <input class="custom-input" type="text" value={subnet} onChange={(e) => {
                       setSubnet(e.target.value);
                       setFee(null);                            
                       }} />
						<button class="custom-button" type="button" onClick={GetFeeFromAPI}>
          		Get staking fee
        		</button>
          </label>
          {error && <p style={{ color: 'red' }}>{error}</p>}
          {fee && <p>The staking fee for subnet {subnet} is: {fee}%</p>}
        </form>
      </div>
    );
};

<StakingFee2 />
