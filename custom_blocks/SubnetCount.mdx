import React, { useState } from 'react';

export const GetSNCount = () => {
    const [subnet, setSubnet] = useState('');
  	const [price, setPrice] = useState('');
  	const [error, setError] = useState(null);

  const GetPriceFromAPI1 = async () => {
			const subnetValue = parseFloat(subnet);

      if (subnetValue) {
        try {
          const response = await fetch(`https://docsdemo.taostats.io/api/dtao/pool/latest/v1?netuid=${subnetValue}`);
          if (!response.ok) {
 					     // Throws an error with the HTTP status text (e.g., "Not Found")
      					throw new Error(`HTTP error! Status: ${response.status}`);
    			}
					const json = await response.json();
          const price = json['data'][0]['price'];
          setPrice(Math.round(price*1000)/1000);
          setError(null);
      	} catch (error) {
        	setError(error.message);
      		}
    	} else {
      	setError('Please enter a valid number');
    	};
  };
    const handleSubnetChange1 = (e) => {
      setSubnet(e.target.value);
    };

    return (
      <div>
        <form onSubmit={(e) => e.preventDefault()}>
          <label>
            Subnet:
            <input type="text" value={subnet} onChange={(e) => setSubnet(e.target.value)} />
						<button type="button" onClick={GetPriceFromAPI1}>
          		Get Price
        		</button>
          </label>
          {error && <p style={{ color: 'red' }}>{error}</p>}
          {price && <p>Price: {price} tao</p>}
        </form>
      </div>
    );
};
