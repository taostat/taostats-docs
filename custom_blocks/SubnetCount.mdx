import React, { useEffect, useState } from 'react';

export const SubnetCount = () => {

    const [error, setError] = useState(null);
    const [subnetCount, setsubnetCount] = useState('');


  const GetCountFromAPI = async () => {

    try {
          const response = await fetch(`https://docsdemo.taostats.io/api/dtao/pool/latest/v1`);
          if (!response.ok) {
            // Throws an error with the HTTP status text (e.g., "Not Found")
            throw new Error(`HTTP error! Status: ${response.status}`);
          }
          const json = await response.json();
          const count = json['pagination']['total_items'] -1;
          setsubnetCount(count);
          setError(null);
        } catch (error) {
          setError(error.message);
        }
  };

    // Runs once when the page (component) loads
  useEffect(() => {
    GetCountFromAPI();
  }, []);
 


    return (
      <div>
          {error && <p style={{ color: 'red' }}>{error}</p>}
          {subnetCount && <p>There are currently {subnetCount} subnets.</p>}
      </div>
    );
};

<SubnetCount />
