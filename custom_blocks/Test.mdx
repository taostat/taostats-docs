import React, { useState } from 'react';

export const ExampleComponent = () => {
    const [alphaIn, setAlphaIn] = useState('');
    const [taoIn, setTaoIn] = useState('');
  	const [price, setPrice] = useState('');

    const calculatePrice = () => {
      const alpha = parseFloat(alphaIn);
      const tao = parseFloat(taoIn);
      if (isNaN(alpha) || isNaN(tao)) {
        setPrice('Error: Invalid input');
      } else {
        const price = tao / alpha;
        setPrice(`Price: ${price}`);
      }
    };



  return (
        <div>
          <h2>Price Calculator</h2>
          <form>
            <label>
              Alpha:
              <input type="number" value={alphaIn} onChange={(e) => setAlphaIn(e.target.value)} />
            </label>
            <br />
            <label>
              Tao:
              <input type="number" value={taoIn} onChange={(e) => setTaoIn(e.target.value)} />
            </label>
            <br />
            <button type="button" onClick={calculatePrice}>
              Calculate Price
            </button>
            <p>Price: {price}</p>
          </form>
        </div>
      );		
};

<ExampleComponent />
