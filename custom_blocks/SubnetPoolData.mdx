import React, { useEffect, useState } from 'react';

export const Subnetpools = () => {

    const [error, setError] = useState(null);
    const [subnets, setSubnets] = useState([]);
		const [subnet, setSubnet] = useState(1);

  const GetSubnetPool = async () => {
					const subnetValue = parseFloat(subnet);

      if (subnetValue) {
      	try {
          //get the top pool data

            const response = await fetch(`https://docsdemo.taostats.io/api/dtao/pool/latest/v1?netuid=${subnetValue}&page=1`);
            if (!response.ok) {
                 // Throws an error with the HTTP status text (e.g., "Not Found")
                  throw new Error(`HTTP error! Status: ${response.status}`);
            }
            const json = await response.json();
            const subnetData = json['data'][0];
						const tao_in = parseFloat((parseFloat(subnetData.total_tao)/1e9).toFixed(3));
          const alpha_in = parseFloat((parseFloat(subnetData.alpha_in_pool)/1e9).toFixed(3));
					const tao_pct = (tao_in/(tao_in+alpha_in)*100).toFixed(2);

          const poolData = {
            "tao_in": tao_in,
            "alpha_in":  alpha_in,
            "tao_pct": parseFloat(tao_pct),
						"alpha_pct": parseFloat((alpha_in/(tao_in+alpha_in)*100).toFixed(2))
          };

            setSubnets(poolData);
            console.log(poolData);
            setError(null);
          } catch (error) {
            setError(error.message);
            }
      } else {
      	setError('Please enter a valid number');
    	};

  };
  
    // Runs once when the page (component) loads
  useEffect(() => {
    GetSubnetPool();
  }, []);


  return (

      <div>
      <div>
        <p style={{ fontFamily: "Georgia, serif", fontSize: 25 }}> Live Query of Subnet Pools</p>
        <p style={{ fontFamily: "Georgia, serif", fontSize: 12 }}> Choose a Subnet to see the breakdown of the subnet pool</p>


        <form onSubmit={(e) => e.preventDefault()}>
          <label>
            Subnet:
            <input type="text" value ={subnet} onChange={(e) => setSubnet(e.target.value)} />
						<button class="custom-button" type="button" onClick={GetSubnetPool}>
          		Get Pool data
        		</button>
          </label>
          {error && <p style={{ color: 'red' }}>{error}</p>}
          {subnets  &&  <p>
 
            <p>tao in pool: {(subnets?.tao_in?.toLocaleString(undefined, { minimumFractionDigits: 3, maximumFractionDigits: 3 }))} tao</p>
            <p>alpha in pool: {(subnets?.alpha_in?.toLocaleString(undefined, { minimumFractionDigits: 3, maximumFractionDigits: 3 }))} alpha </p>
            <p>% tao: {subnets.tao_pct}% </p>
						<p>% alpha: {subnets.alpha_pct}%  </p>



            </p>}



          {error && <p style={{ color: 'red' }}>{error}</p>}
          <label>
					<p style={{ fontFamily: "Georgia, serif", fontSize: 10 }}> Data from Taostats API</p>
          </label>


         




        </form>
      </div>
  
      </div>
    );
};

<Subnetpools />
