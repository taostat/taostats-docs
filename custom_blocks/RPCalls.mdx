import React, { useEffect, useState } from "react";

export const SearchPallets = () => {
  const [pallets, setPallets] = useState(null);
  const [selectedPallet, setSelectedPallet] = useState("");
  const [selectedMethod, setSelectedMethod] = useState("");
  const [error, setError] = useState(null);

  const GetPalletsFromAPI = async () => {
    try {
      const response = await fetch(
        `https://docsdemo.taostats.io/api/rpc/list?network=lite`
      );
      if (!response.ok) {
        throw new Error(`HTTP error! Status: ${response.status}`);
      }
      const data = await response.json();
      console.log("RPC Response:", data);

      setPallets(data);

      // If SubtensorModule exists, select its first method by default
      if (data["subtensorModule"]) {
        console.log("here");
        setSelectedPallet('subtensorModule');
      } else {
        const palletNames = Object.keys(data);
        if (palletNames.length > 0) {
          if (palletNames.length > 0) setSelectedPallet(palletNames[0]);
        }
      }
    } catch (error) {
      setError(error.message);
    }
  }; 

  useEffect(() => {
    GetPalletsFromAPI();
  }, []);
  

  // 👇 This effect runs every time pallets OR selectedPallet changes
  useEffect(() => {
    if (pallets && selectedPallet && pallets[selectedPallet]) {
      const methods = Object.keys(pallets[selectedPallet]);
      setSelectedMethod(methods[0] || "");
    }
  }, [pallets, selectedPallet]); // ✅ runs after pallets+selectedPallet update

  const handlePalletChange = (e) => setSelectedPallet(e.target.value);
  const handleMethodChange = (e) => setSelectedMethod(e.target.value);

  // ✅ Safe checks before using Object.keys
  const palletNames = pallets ? Object.keys(pallets) : [];
  const methodNames =
    pallets && pallets[selectedPallet]
      ? Object.keys(pallets[selectedPallet])
      : [];

  return (
    <div>
      {error && <p style={{ color: "red" }}>Error: {error}</p>}
      {!pallets ? (
        <p>Loading...</p>
      ) : (
        <form style={{ marginTop: "1em" }}>
          {/* Pallet Dropdown */}
          <div style={{ marginBottom: "1em" }}>
            <label htmlFor="pallet">Pallet: </label>
            <select
              id="pallet"
              value={selectedPallet}
              onChange={handlePalletChange}
            >
              {palletNames.map((pallet) => (
                <option key={pallet} value={pallet}>
                  {pallet}
                </option>
              ))}
            </select>
          </div>

          {/* Method Dropdown */}
          <div>
            <label htmlFor="method">Method: </label>
            <select
              id="method"
              value={selectedMethod}
              onChange={handleMethodChange}
            >
              {methodNames.map((method) => (
                <option key={method} value={method}>
                  {method}
                </option>
              ))}
            </select>
          </div>
        </form>
      )}
    </div>
  );
};


<SearchPallets />;
