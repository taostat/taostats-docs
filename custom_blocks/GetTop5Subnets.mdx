import React, { useState } from 'react';

export const Top5Subnets = () => {
    const [subnet, setSubnet] = useState('');
  	const [fee, setFee] = useState('');
    const [error, setError] = useState(null);
    const [subnets, setSubnets] = useState(null);


  const GetTopSubnets = async () => {

        try {
          const response = await fetch(`https://docsdemo.taostats.io/api/subnet/latest/v1?limit=5&order=emission_desc`);
          if (!response.ok) {
 					     // Throws an error with the HTTP status text (e.g., "Not Found")
      					throw new Error(`HTTP error! Status: ${response.status}`);
    			}
          const json = await response.json();
          const subnetData = json['data'];
          const topSubnets = subnetData.map(subnet => {
            return{
              hotkey: subnet.owner_hotkey.ss58,
              netuid: subnet.netuid,
							emission: parseFloat(subnet.emission)/1e9
            };
          });
          setSubnets(topSubnets);

          setError(null);
      	} catch (error) {
        	setError(error.message);
      		}

  };
    const handleSubnetChange = (e) => {
      setSubnet(e.target.value);
    };

    return (
      <div>
        <form onSubmit={(e) => e.preventDefault()}>
          <label>
            Subnet:
            <input type="text" value={subnet} onChange={(e) => {
                       setSubnet(e.target.value);
                       setFee(null);                            
                       }} />
						<button type="button" onClick={GetTopSubnets}>
          		Get staking fee
        		</button>
          </label>
          {error && <p style={{ color: 'red' }}>{error}</p>}
          {subnets && <p>The staking fee for subnet {subnets} </p>}
        </form>
      </div>
    );
};

<Top5Subnets />
