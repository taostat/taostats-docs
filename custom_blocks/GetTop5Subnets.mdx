import React, { useEffect, useState } from 'react';

export const Top5Subnets = () => {

    const [error, setError] = useState(null);
    const [subnets, setSubnets] = useState([]);


  const GetTopSubnets = async () => {

    try {
				//get the top subnets

          const response = await fetch(`https://docsdemo.taostats.io/api/subnet/latest/v1?limit=5&order=emission_desc`);
          if (!response.ok) {
 					     // Throws an error with the HTTP status text (e.g., "Not Found")
      					throw new Error(`HTTP error! Status: ${response.status}`);
    			}
          const json = await response.json();
      		const subnetData = json['data'];
	        //get the identities
          //get the identities
              const response1 = await fetch(`https://docsdemo.taostats.io/api/subnet/identity/v1`);
              if (!response1.ok) {
                   // Throws an error with the HTTP status text (e.g., "Not Found")
                    throw new Error(`HTTP error! Status: ${response1.status}`);
              }
          const identityJson = await response1.json();
					const subnetIdentities = identityJson['data']

          const topSubnets = subnetData.map(subnet => {
            // find the matching identity by netuid
  					const identity = subnetIdentities.find(item => item.netuid === subnet.netuid);
						const snEmission = (parseFloat(subnet.emission)/1e9*100).toFixed(3);
            return{
              hotkey: subnet.owner_hotkey.ss58,
              netuid: subnet.netuid,
              emission: snEmission,
              subnet_name: identity ? identity.subnet_name : null, // fallback if no match
							subnet_logo: identity?.logo_url ? identity.logo_url: 'https://taostats.io/images/subnets/0.webp?w=64&q=75' // fallback if no match


            };
          });
          setSubnets(topSubnets);
					console.log(topSubnets);
          setError(null);
      	} catch (error) {
        	setError(error.message);
      		}

  };
  
    // Runs once when the page (component) loads
  useEffect(() => {
    GetTopSubnets();
  }, []);


  return (

      <div>

          <label>

					<p style={{ fontFamily: "Georgia, serif", fontSize: 25 }}> Live View of Top Subnets</p>
          </label>
				  <table border="1" id="subnet-table" className="custom-table" >
            <thead>
              <tr style={{backgroundColor:'#6b6b6b', color: '#09d9ba',textAlign:'center' }}>
                <th style = {{maxWidth: 20}}>Logo</th>
								<th style = {{maxWidth: 40}}>Name</th>
                <th style = {{maxWidth: 20}}>NetUID</th>
                <th style = {{maxWidth: 50}}>Emission</th>
              </tr>
            </thead>
            <tbody>
          {subnets.map(subnet => (
            <tr key={subnet.netuid} style={{backgroundColor:'#6b6b6b', color: '#ebc249', textAlign:'center'}}>
              <td style = {{maxWidth: 20}}>
                <img
                  src={subnet.subnet_logo}
                  alt={subnet.subnet_name}
                  width={32}
                  height={32}
                />
              </td>
              <td style = {{maxWidth: 40}}>{subnet.subnet_name}</td>
              <td style = {{maxWidth: 20}}>{subnet.netuid}</td>
      
              <td style = {{maxWidth: 50}}>{subnet.emission}%</td>
            </tr>
          ))}
        </tbody>
          </table>


          {error && <p style={{ color: 'red' }}>{error}</p>}
          <label>
					<p style={{ fontFamily: "Georgia, serif", fontSize: 10 }}> Data from Taostats API</p>
          </label>

      </div>
    );
};

<Top5Subnets />
