import React, { useState } from 'react';

export const Top5Subnets = () => {

    const [error, setError] = useState(null);
    const [subnets, setSubnets] = useState(null);


  const GetTopSubnets = async () => {

    try {
				//get the top subnets

          const response = await fetch(`https://docsdemo.taostats.io/api/subnet/latest/v1?limit=5&order=emission_desc`);
          if (!response.ok) {
 					     // Throws an error with the HTTP status text (e.g., "Not Found")
      					throw new Error(`HTTP error! Status: ${response.status}`);
    			}
          const json = await response.json();
      		const subnetData = json['data'];
	        //get the identities
          //get the identities
              const response1 = await fetch(`https://docsdemo.taostats.io/api/subnet/identity/v1`);
              if (!response1.ok) {
                   // Throws an error with the HTTP status text (e.g., "Not Found")
                    throw new Error(`HTTP error! Status: ${response1.status}`);
              }
          const identityJson = await response1.json();
					const subnetIdentities = identityJson['data']

          const topSubnets = subnetData.map(subnet => {
            // find the matching identity by netuid
  					const identity = subnetIdentities.find(item => item.netuid === subnet.netuid);
	
            return{
              hotkey: subnet.owner_hotkey.ss58,
              netuid: subnet.netuid,
              emission: parseFloat(subnet.emission)/1e9,
              subnet_name: identity ? identity.subnet_name : null, // fallback if no match
							subnet_logo: identity ? identity.logo_url : 'https://taostats.io/images/subnets/0.webp?w=64&q=75' // fallback if no match


            };
          });
          setSubnets(topSubnets);
					console.log(topSubnets);
          setError(null);
      	} catch (error) {
        	setError(error.message);
      		}

  };


  return (

      <div>
        <form onSubmit={(e) => e.preventDefault()}>
          <label>

						<button type="button" onClick={GetTopSubnets}>
          		Get top subnets
        		</button>
          </label>
				  <table border="1" id="subnet-table" style={{backgroundColor:'green', color:'black' }}>
            <thead>
              <tr style={{backgroundColor:'#6b6b6b', color: '#09d9ba' }}>
                <th>Name</th>
                <th>NetUID</th>
                <th>Hotkey</th>
                <th>Emission</th>
              </tr>
            </thead>
            <tbody></tbody>
          </table>

          {error && <p style={{ color: 'red' }}>{error}</p>}
          {subnets && <p>The staking fee for subnet {subnets} </p>}
        </form>
      </div>
    );
};

<Top5Subnets />
