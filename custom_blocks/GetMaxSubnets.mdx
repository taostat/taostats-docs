import React, { useState, useEffect } from 'react';

export const GetMaxSubnets = () => {
  	const [pallets, setPallets] = useState(null);
  	const [error, setError] = useState(null);

  const GetSubnetLimit = async () => {

        try {
          const response = await fetch(`https://docsdemo.taostats.io/api/rpc/query?network=lite`,{
            method: "POST",
						headers: {
		            "Content-Type": "application/json"
          	},
	          body: JSON.stringify({
              jsonrpc: "2.0",
              id: 1,
              module: "subtensorModule",
						  storage_function: "subnetLimit",
              params: []
            })


					});
          if (!response.ok) {
 					     // Throws an error with the HTTP status text (e.g., "Not Found")
      					throw new Error(`HTTP erjror! Status: ${response.status}`);
    			}
					const json = await response.json();
      		console.log("RPC Response:", json); // helpful for debugging

          // Example: If json.result.methods exists and is an array, do something
          
          setPallets(json.value);
					console.log(pallets);
          setError(null);


      	} catch (error) {
        	setError(error.message);
      		}

  };
    // Runs once when the page (component) loads
    // Runs once when the p age (component) loads
  useEffect(() => {
    // Update token based on direction
  GetSubnetLimit();
		
  }, []);


    return (
      <div>
        <form onSubmit={(e) => e.preventDefault()}>
          <label>


          </label>
          {error && <p style={{ color: 'red' }}>{error}</p>}
          {pallets !== null ? <p>The current Subnet Limit is: {pallets}</p> : <p>Loading...</p>}

        </form>
      </div>
    );
};

<GetMaxSubnets />
