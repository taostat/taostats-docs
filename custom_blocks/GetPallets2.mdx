import React, { useState } from "react";

export const GetPallets2 = () => {
  const [pallets, setPallets] = useState(null);
  const [error, setError] = useState(null);

  const GetPalletsFromAPI = async () => {
    try {
      const response = await fetch(
        `https://docsdemo.taostats.io/api/rpc/list?network=lite`
      );
      if (!response.ok) throw new Error(`HTTP error! Status: ${response.status}`);

      const json = await response.json();
      console.log("RPC Response:", json);

      setPallets(json);
      setError(null);
    } catch (err) {
      setError(err.message);
    }
  };

  return (
    <div>
      <button type="button" onClick={GetPalletsFromAPI}>
        Get Latest RPC Methods
      </button>

      {error && <p style={{ color: "red" }}>{error}</p>}

      {pallets && (
        <div style={{ marginTop: "1em" }}>
          {Object.entries(pallets).map(([pallet, methods]) => (
            <div key={pallet} style={{ marginBottom: "1em" }}>
              <h3>{pallet}</h3>
              <p>{methods.join(", ")}</p>
            </div>
          ))}
        </div>
      )}
    </div>
  );
};
