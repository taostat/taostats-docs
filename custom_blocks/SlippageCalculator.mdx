import React, { useEffect, useState } from 'react';

export const SlippageCalculator = () => {

    const [error, setError] = useState(null);
    const [subnets, setSubnets] = useState([]);
    const [subnet, setSubnet] = useState(1);
    const [amount, setAmount] = useState(10);
	  const [direction, setDirection] = useState('tao_to_alpha');
		const [token, setToken] = useState('tao');


  const Slippage = async () => {
					const subnetValue = parseFloat(subnet);

      if (subnetValue) {
      	try {
          //get the top pool data
          const amount_rao = amount*1e9;
          const token = "tao";
          if ({direction} == "alpha_to_tao"){setToken("tao");}
					else {setToken("alpha")}


            const response = await fetch(`https://docsdemo.taostats.io/api/dtao/slippage/v1?netuid=${subnet}&input_tokens=${amount_rao}&direction=${direction}`);
            if (!response.ok) {
                 // Throws an error with the HTTP status text (e.g., "Not Found")
                  throw new Error(`HTTP error! Status: ${response.status}`);
            }
            const json = await response.json();
            const subnetData = json['data'][0];
						const price = parseFloat((parseFloat(subnetData.alpha_price)).toFixed(5));
            const output_tokens = parseFloat((parseFloat(subnetData.output_tokens)/1e9).toFixed(5));
            const expected_tokens = parseFloat((parseFloat(subnetData.expected_output_tokens)/1e9).toFixed(5));
					  const slippage = parseFloat((parseFloat(subnetData.slippage)).toFixed(10));


          const poolData = {
            "price": price,
            "output_tokens":  output_tokens,
            "expected_tokens": expected_tokens,
            "slippage": slippage
          };

            setSubnets(poolData);
            console.log(poolData);
            setError(null);
          } catch (error) {
            setError(error.message);
            }
      } else {
      	setError('Please enter a valid number');
    	};

  };
  
    // Runs once when the page (component) loads
  useEffect(() => {
    Slippage();
  }, []);


  return (

      <div>
      <div>
        <p style={{ fontFamily: "Georgia, serif", fontSize: 25 }}> Live Subnet Slippage Calculator</p>
        <p style={{ fontFamily: "Georgia, serif", fontSize: 12 }}> Choose a Subnet to see the breakdown of the subnet pool</p>


        <form onSubmit={(e) => e.preventDefault()}>
          <label>
           <p> Subnet:
            <input class="custom-input" type="text" value ={subnet} onChange={(e) => setSubnet(e.target.value)} /></p>
            <p> Amount:
              <input class="custom-input" type="text" value ={amount} onChange={(e) => setAmount(e.target.value)} />
						<select name="direction" id="direction">
              <option value ="tao_to_alpha" onChange={(e) => setdirection(e.target.value)}>tao to alpha</option>
							<option value ="alpha_to_tao" onChange={(e) => setdirection(e.target.value)}>alpha to tao</option>
              </select></p>
            

						<button class="custom-button" type="button" onClick={Slippage}>
          		Get Slippage estimate
        		</button>
          </label>
          {error && <p style={{ color: 'red' }}>{error}</p>}
          {subnets  &&  <p>
 						<p/><p/>
            <p>tokens Expected: {(subnets?.expected_tokens?.toLocaleString(undefined, { minimumFractionDigits: 3, maximumFractionDigits: 3 }))} {token}</p>
            <p>tokens received: {(subnets?.output_tokens?.toLocaleString(undefined, { minimumFractionDigits: 3, maximumFractionDigits: 3 }))}  {token}</p>
            <p>Slippage: {subnets.slippage}% </p>
		



            </p>}



          {error && <p style={{ color: 'red' }}>{error}</p>}
          <label>
					<p style={{ fontFamily: "Georgia, serif", fontSize: 10 }}> Data from Taostats API</p>
          </label>


         




        </form>
      </div>
  
      </div>
    );
};

<SlippageCalculator />
